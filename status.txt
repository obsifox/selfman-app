# ============================================================
#  راهنمای کنترل وضعیت برنامهٔ سلف‌من  (فقط برای مالک برنامه)
#  خط‌هایی که با # شروع می‌شوند فقط توضیح‌اند؛ برنامه آن‌ها را
#  نمی‌خواند. برای کنترل برنامه فقط مقادیر پایین را عوض کنید.
# ------------------------------------------------------------
#  version : آخرین نسخهٔ منتشرشدهٔ برنامه (مثل 2.73)
#  request : شرط درخواست آپدیت؛ هر نسخه‌ای که این شرط را داشته
#            باشد موقع ورود «می‌خوای آپدیت کنی؟» می‌بیند (قابل رد).
#            مثال:  request = <2.74  یعنی پایین‌تر از 2.73
#  state   : وضعیت برنامه، یکی از این حرف‌ها:
#            r = عادی (ورود آزاد؛ فقط شرط request اعمال می‌شود)
#            u = آپدیت اجباری؛ نسخه‌های پایین‌تر از version مجبور
#                به نصب می‌شوند، مگر نسخه‌شان در support باشد که
#                فقط از آن‌ها خواسته می‌شود
#            s = تعلیق موقت؛ هیچ‌کس وارد نمی‌شود
#            c = بسته؛ هیچ‌کس وارد نمی‌شود
#            p = نیاز به پرداخت/فعال‌سازی (لینک را در url بگذارید)
#  support : نسخه‌های قدیمیِ هنوز پشتیبانی‌شده، با ویرگول جدا
#            (در state=u این‌ها فقط پیام آپدیت می‌بینند)
#  message : (اختیاری) متن دلخواهی که در صفحهٔ وضعیت نشان داده
#            می‌شود؛ اگر خالی باشد متن پیش‌فرض می‌آید
#  url     : (اختیاری) لینک پرداخت/پشتیبانی برای state=p
# ------------------------------------------------------------
#  نکته: اعداد و حرف‌های لاتین را همین‌طور نگه دارید؛ بعد از
#  هر ویرایش چند دقیقه صبر کنید و برنامه را کامل ببندید و باز
#  کنید (برنامه موقع ورود و هنگام برگشتن از پس‌زمینه وضعیت را
#  دوباره می‌خواند). برای انتشار نسخهٔ جدید، این فایل به‌تنهایی
#  کافی نیست؛ باید ریلیز همان نسخه هم در گیت‌هاب ساخته شود.
# ============================================================

request = <2.74
state = r
support = 2.73
